cmake_minimum_required(VERSION 3.30)
project(KaGuYa LANGUAGES C CXX RC)

if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang"
   OR NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  message(FATAL_ERROR "Use the clang-cl-x86 toolchain preset; no implicit compiler fallback.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "KaGuYa requires x86 pointers and the original 32-bit ABI.")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
  message(FATAL_ERROR "Choose the supplied Debug or Release preset.")
endif()
set(_kaguya_logging_default OFF)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(_kaguya_logging_default ON)
endif()
option(KAGUYA_ENABLE_LOGGING "Compile the file logging backend into EXE and DLLs" ${_kaguya_logging_default})
set(_kaguya_logging_value 0)
if(KAGUYA_ENABLE_LOGGING)
  set(_kaguya_logging_value 1)
endif()

set(KAGUYA_GAME "mamatomo" CACHE STRING "Game resource profile compiled into Start")
# 清除旧Size试验的缓存开关，Release固定使用小体积配置和统一产物目录。
unset(KAGUYA_ARTIFACT_DIR CACHE)
unset(KAGUYA_OPTIMIZE_SIZE CACHE)
set(KAGUYA_ARTIFACT_DIR "${PROJECT_SOURCE_DIR}/Release")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  string(APPEND KAGUYA_ARTIFACT_DIR "/Debug")
endif()
if(CMAKE_CXX_STANDARD_LATEST LESS 26)
  message(FATAL_ERROR "The LLVM toolchain must support the current C++ latest mode.")
endif()
set(CMAKE_CXX_STANDARD "${CMAKE_CXX_STANDARD_LATEST}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

# 原Release未定义NDEBUG；异常模式由每个求值后的源文件独立设置。
foreach(_language C CXX)
  string(REPLACE "/EHsc" "" CMAKE_${_language}_FLAGS "${CMAKE_${_language}_FLAGS}")
  string(REPLACE "/DNDEBUG" "" CMAKE_${_language}_FLAGS_RELEASE "${CMAKE_${_language}_FLAGS_RELEASE}")
  string(REPLACE "/RTC1" "" CMAKE_${_language}_FLAGS_DEBUG "${CMAKE_${_language}_FLAGS_DEBUG}")
  string(REPLACE "/ZI" "/Zi" CMAKE_${_language}_FLAGS_DEBUG "${CMAKE_${_language}_FLAGS_DEBUG}")
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    string(REPLACE "/O2" "/O1" CMAKE_${_language}_FLAGS_RELEASE "${CMAKE_${_language}_FLAGS_RELEASE}")
    string(REPLACE "/Ob2" "/Ob1" CMAKE_${_language}_FLAGS_RELEASE "${CMAKE_${_language}_FLAGS_RELEASE}")
  endif()
endforeach()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)

set(KAGUYA_MODEL_DIR "${CMAKE_BINARY_DIR}/generated/msbuild")
include(cmake/ImportVcxproj.cmake)
kaguya_import_projects()
include("${KAGUYA_MODEL_DIR}/targets.cmake")

# 构建工具不发布到Release；每个Start链接成功后由它核对资源元数据。
add_executable(KaGuYaResourceMetadata EXCLUDE_FROM_ALL cmake/native/ResourceMetadata.cpp)
set_target_properties(KaGuYaResourceMetadata PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts/tools"
  PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts/symbols/ResourceMetadata")
target_compile_options(KaGuYaResourceMetadata PRIVATE /W4 /EHsc /utf-8)
include(cmake/ResourceMetadata.cmake)
target_compile_definitions(Start PRIVATE KAGUYA_GAME_NUMBER=18)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  # 保留导出和强制CRT初始化根；仅移除未引用COMDAT，不合并函数地址。
  foreach(_target Start Graphics RenderDX Sound)
    target_link_options(${_target} PRIVATE /OPT:REF /OPT:NOICF)
  endforeach()
endif()

include(cmake/SoundCompatibility.cmake)
kaguya_register_sound_compatibility()
include(cmake/GameTargets.cmake)
kaguya_register_game_targets()

set(_license_dir "${KAGUYA_ARTIFACT_DIR}/licenses")
file(MAKE_DIRECTORY "${_license_dir}")
configure_file(3rdParty/Xiph/libogg-1.3.6/COPYING "${_license_dir}/libogg-1.3.6.txt" COPYONLY)
configure_file(3rdParty/Xiph/libvorbis-1.3.7/COPYING "${_license_dir}/libvorbis-1.3.7.txt" COPYONLY)
configure_file(3rdParty/Spdlog/LICENSE "${_license_dir}/spdlog.txt" COPYONLY)
configure_file(3rdParty/Spdlog/include/spdlog/fmt/bundled/fmt.license.rst "${_license_dir}/fmt.txt" COPYONLY)

message(STATUS "KaGuYa: x86 clang-cl / lld-link / C++ latest; MSVC STL/CRT/ATL ${KAGUYA_MSVC_VERSION}")
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  message(STATUS "Release: /O1 /Ob1 /OPT:REF /OPT:NOICF -> ${KAGUYA_ARTIFACT_DIR}")
endif()
message(STATUS "Compilation database: ${CMAKE_BINARY_DIR}/compile_commands.json")
message(STATUS "File logging backend: ${KAGUYA_ENABLE_LOGGING}")
